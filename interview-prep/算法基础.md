## 一、排序算法 ⭐⭐⭐⭐⭐

### 1.1 快速排序

```js
// 原地快排（推荐手写这个版本）
function quickSort(arr, left = 0, right = arr.length - 1) {
  if (left >= right) return arr

  const pivot = partition(arr, left, right)
  quickSort(arr, left, pivot - 1)
  quickSort(arr, pivot + 1, right)
  return arr
}

function partition(arr, left, right) {
  const pivot = arr[right]  // 选最右为基准
  let i = left               // i 指向第一个大于 pivot 的位置

  for (let j = left; j < right; j++) {
    if (arr[j] < pivot) {
      [arr[i], arr[j]] = [arr[j], arr[i]]
      i++
    }
  }
  [arr[i], arr[right]] = [arr[right], arr[i]]
  return i
}

// 时间复杂度：平均 O(n log n)，最坏 O(n²)
// 空间复杂度：O(log n)（递归栈）
// 不稳定排序
```

### 1.2 归并排序

```js
function mergeSort(arr) {
  if (arr.length <= 1) return arr

  const mid = Math.floor(arr.length / 2)
  const left = mergeSort(arr.slice(0, mid))
  const right = mergeSort(arr.slice(mid))

  return merge(left, right)
}

function merge(left, right) {
  const result = []
  let i = 0, j = 0

  while (i < left.length && j < right.length) {
    if (left[i] < right[j]) {
      result.push(left[i++])
    } else {
      result.push(right[j++])
    }
  }
  return [...result, ...left.slice(i), ...right.slice(j)]
}

// 时间复杂度：O(n log n)，稳定
// 空间复杂度：O(n)
// 稳定排序
```

### 1.3 排序算法对比

| 算法 | 平均时间 | 最坏时间 | 空间 | 稳定 |
|------|---------|---------|------|------|
| 冒泡 | O(n²) | O(n²) | O(1) | 是 |
| 选择 | O(n²) | O(n²) | O(1) | 否 |
| 插入 | O(n²) | O(n²) | O(1) | 是 |
| 快速 | O(n log n) | O(n²) | O(log n) | 否 |
| 归并 | O(n log n) | O(n log n) | O(n) | 是 |
| 堆排 | O(n log n) | O(n log n) | O(1) | 否 |

## 二、树遍历 ⭐⭐⭐⭐

### 2.1 二叉树遍历

```js
// 递归版（面试先写这个）
// 前序：根 → 左 → 右
function preorder(root) {
  if (!root) return []
  return [root.val, ...preorder(root.left), ...preorder(root.right)]
}

// 中序：左 → 根 → 右（BST 中序 = 有序数组）
function inorder(root) {
  if (!root) return []
  return [...inorder(root.left), root.val, ...inorder(root.right)]
}

// 后序：左 → 右 → 根
function postorder(root) {
  if (!root) return []
  return [...postorder(root.left), ...postorder(root.right), root.val]
}
```

```js
// 迭代版（面试加分项）
// 中序迭代：左 → 根 → 右
function inorderIterative(root) {
  const result = []
  const stack = []
  let curr = root

  while (curr || stack.length) {
    while (curr) {        // 一路向左
      stack.push(curr)
      curr = curr.left
    }
    curr = stack.pop()    // 弹出并访问
    result.push(curr.val)
    curr = curr.right     // 转向右子树
  }
  return result
}

// 前序迭代：根 → 左 → 右
function preorderIterative(root) {
  if (!root) return []
  const result = []
  const stack = [root]

  while (stack.length) {
    const node = stack.pop()
    result.push(node.val)
    if (node.right) stack.push(node.right) // 右先入栈
    if (node.left) stack.push(node.left)   // 左后入栈（先出）
  }
  return result
}
```

### 2.2 层序遍历 (BFS)

```js
function levelOrder(root) {
  if (!root) return []
  const result = []
  const queue = [root]

  while (queue.length) {
    const level = []
    const levelSize = queue.length

    for (let i = 0; i < levelSize; i++) {
      const node = queue.shift()
      level.push(node.val)
      if (node.left) queue.push(node.left)
      if (node.right) queue.push(node.right)
    }
    result.push(level)
  }
  return result
}
```

## 三、链表操作 ⭐⭐⭐⭐

### 3.1 反转链表

```js
// 迭代（推荐）
function reverseList(head) {
  let prev = null
  let curr = head
  while (curr) {
    const next = curr.next  // 保存下一个
    curr.next = prev         // 反转指针
    prev = curr              // 前移
    curr = next              // 前移
  }
  return prev
}

// 递归
function reverseList(head) {
  if (!head || !head.next) return head
  const newHead = reverseList(head.next)
  head.next.next = head
  head.next = null
  return newHead
}
```

### 3.2 检测环

```js
// 快慢指针
function hasCycle(head) {
  let slow = head, fast = head
  while (fast && fast.next) {
    slow = slow.next
    fast = fast.next.next
    if (slow === fast) return true
  }
  return false
}
```

### 3.3 合并两个有序链表

```js
function mergeTwoLists(l1, l2) {
  const dummy = { next: null }
  let curr = dummy

  while (l1 && l2) {
    if (l1.val < l2.val) {
      curr.next = l1
      l1 = l1.next
    } else {
      curr.next = l2
      l2 = l2.next
    }
    curr = curr.next
  }
  curr.next = l1 || l2
  return dummy.next
}
```

## 四、动态规划入门 ⭐⭐⭐

### 4.1 爬楼梯 (LeetCode 70)

```js
// dp[i] = dp[i-1] + dp[i-2]  （斐波那契）
function climbStairs(n) {
  if (n <= 2) return n
  let a = 1, b = 2
  for (let i = 3; i <= n; i++) {
    [a, b] = [b, a + b]
  }
  return b
}
```

### 4.2 最大子数组和 (LeetCode 53)

```js
// dp[i] = max(nums[i], dp[i-1] + nums[i])
function maxSubArray(nums) {
  let max = nums[0]
  let pre = nums[0]
  for (let i = 1; i < nums.length; i++) {
    pre = Math.max(nums[i], pre + nums[i])
    max = Math.max(max, pre)
  }
  return max
}
```

## 五、字符串 ⭐⭐⭐

### 5.1 最长无重复子串 (LeetCode 3)

```js
// 滑动窗口
function lengthOfLongestSubstring(s) {
  const map = new Map()
  let left = 0, max = 0

  for (let right = 0; right < s.length; right++) {
    if (map.has(s[right])) {
      left = Math.max(left, map.get(s[right]) + 1)
    }
    map.set(s[right], right)
    max = Math.max(max, right - left + 1)
  }
  return max
}
```

### 5.2 有效的括号 (LeetCode 20)

```js
function isValid(s) {
  const stack = []
  const map = { ')': '(', ']': '[', '}': '{' }

  for (const ch of s) {
    if (!map[ch]) {
      stack.push(ch)
    } else if (stack.pop() !== map[ch]) {
      return false
    }
  }
  return stack.length === 0
}
```

## 六、经典场景题 ⭐⭐⭐⭐

### 6.1 数组转树

```js
// 给出 [{id, parentId, name}] 转为嵌套树结构
function arrayToTree(items) {
  const map = {}
  const roots = []

  // 先建 map
  items.forEach(item => {
    map[item.id] = { ...item, children: [] }
  })

  // 挂载父子关系
  items.forEach(item => {
    const node = map[item.id]
    if (item.parentId === 0 || !map[item.parentId]) {
      roots.push(node)
    } else {
      map[item.parentId].children.push(node)
    }
  })

  return roots
}
```

### 6.2 树转数组

```js
function treeToArray(tree) {
  const result = []
  const stack = [...tree]

  while (stack.length) {
    const node = stack.pop()
    const { children, ...rest } = node
    result.push(rest)
    if (children && children.length) {
      stack.push(...children)
    }
  }
  return result
}
```

## 七、面试策略

前端算法面试的关键：
1. 先和面试官确认理解：输入输出、边界条件
