FSM 核心实现（转移表 + Guard，Demo 里跑的就是这段）
// 转移表：state + event → { target, guard }。 没有匹配项 = 非法转移，直接拒绝
const transitions = {
  IDLE:             { START:           { target: 'COLLECTING' } },
  COLLECTING:       { SLOT_UPDATE:     { target: 'COLLECTING' },  // 自环：填槽不换状态
                      SLOTS_COMPLETE:  { target: 'VALIDATING', guard: allSlotsFilled },
                      CANCEL:          { target: 'CANCELLED' } },
  VALIDATING:       { BALANCE_OK:      { target: 'PENDING_CONFIRM' },
                      BALANCE_LACK:    { target: 'COLLECTING' } },
  PENDING_CONFIRM:  { CONFIRM:         { target: 'SUBMITTED' },
                      EDIT:            { target: 'COLLECTING' },
                      CANCEL:          { target: 'CANCELLED' } },
};

function transition(state, event, ctx) {
  const rule = transitions[state]?.[event];
  if (!rule)                return { ok: false, reason: `非法转移：${state} 不响应 ${event}` };
  if (rule.guard && !rule.guard(ctx))
                          return { ok: false, reason: 'Guard 拦截：条件不满足' };
  return { ok: true, target: rule.target };  // 通过才允许换状态
}
编排视角：LLM 的产出被降级为「事件 + 槽位数据」——它可以提议 CONFIRM，但能不能发生由转移表说了算。 这就是「代码态编排」和「提示词编排」的本质区别：前者把纪律编译进控制流，后者把纪律写成自然语言祈祷模型遵守。
