核心实现（简化版）
// 全局依赖容器
const targetMap = new WeakMap();
let activeEffect = null;

function track(target, key) {
  // state.count 被读取 → 把当前 effect 记入依赖
  getDep(target, key).add(activeEffect);
}

function trigger(target, key) {
  // state.count 被修改 → 取出依赖并重新执行
  const dep = targetMap.get(target)?.get(key);
  if (dep) [...dep].forEach(fn => fn());
}

function reactive(target) {
  return new Proxy(target, {
    get(target, key, receiver) {
      track(target, key);          // 读取时收集
      return Reflect.get(target, key, receiver);
    },
    set(target, key, value, receiver) {
      const old = target[key];
      const ok = Reflect.set(target, key, value, receiver);
      if (old !== value) trigger(target, key); // 写入时触发
      return ok;
    },
    deleteProperty(target, key) {
      const had = hasOwn(target, key);
      const ok = Reflect.deleteProperty(target, key);
      if (had && ok) trigger(target, key); // 删除也能感知
      return ok;
    }
  });
}
